int potenciometro = A1;
int botao = 7;
int led6 = 6;
int led5 = 5;
int estadoLed = LOW;
int estadoAnterior = LOW;

void setup()
{
  Serial.begin(9600);
  pinMode(botao, INPUT);
  pinMode(led6, OUTPUT);
  pinMode(led5, OUTPUT);
}

void loop()
{

  int valor = analogRead(potenciometro);
              
  int valorPWM = map(valor, 0, 1023, 0, 255);
               
  analogWrite(led6, valorPWM);
  delay(10);
  
  int leituraAtual = digitalRead(botao);

  if(leituraAtual == HIGH && estadoAnterior == LOW){
  	estadoLed = !estadoLed;
    delay(50);
  }
  digitalWrite(led5, estadoLed);
  estadoAnterior = leituraAtual;
}
